const file = this.files[0]; if (!file) return; console.log( "🎵 GLCTC1 Audio selected:", file.name ); // Clear previous analysis document.getElementById("bpm").value = ""; document.getElementById("key").value = ""; document.getElementById("frequency_hz").value = ""; try { console.log( "🔬 Reading audio file..." ); const arrayBuffer = await file.arrayBuffer(); const AudioContext = window.AudioContext || window.webkitAudioContext; const analyserContext = new AudioContext(); const audioBuffer = await analyserContext.decodeAudioData( arrayBuffer ); console.log( "✅ WAV decoded:", audioBuffer.duration.toFixed(2), "seconds" ); console.log( "Sample rate:", audioBuffer.sampleRate ); console.log( "Channels:", audioBuffer.numberOfChannels ); await analyserContext.close(); console.log( "🎵 Audio is ready for BPM / Key / Frequency analysis." ); } catch (error) { console.error( "❌ GLCTC1 AUDIO ANALYSIS ERROR:", error ); alert( "Unable to analyse this audio file." ); } });
